## 4. CODE BOILERPLATES & WEBMCP INTEGRATION HOOKS

### 4.1 WebMCP Tool Registration Hook (`useWebMCP.ts`)

```typescript
import { useEffect, useState } from "react";

export interface MemoryPayload {
  content: string;
  category: "preference" | "workflow" | "project" | "constraint" | "other";
  tags: string[];
  source: string;
  confidence: number;
}

export function useWebMCP(sourceName: string, onActivityLog?: (log: string) => void) {
  const [isSupported, setIsSupported] = useState<boolean>(false);

  useEffect(() => {
    // Check for WebMCP availability in navigator or document model context
    const modelContext = (window as any).navigator?.modelContext || (window as any).document?.modelContext;
    
    if (!modelContext || typeof modelContext.registerTool !== "function") {
      console.warn("WebMCP is not available in this browser environment.");
      setIsSupported(false);
      return;
    }

    setIsSupported(true);

    // Register store_context tool
    modelContext.registerTool({
      name: "store_context",
      description: "Persist useful user context into the Tether shared memory layer.",
      inputSchema: {
        type: "object",
        properties: {
          content: { type: "string", description: "The preference or context to remember" },
          category: { 
            type: "string", 
            enum: ["preference", "workflow", "project", "constraint", "other"],
            description: "Category of memory" 
          },
          tags: { type: "array", items: { type: "string" }, description: "Tags for lookup" },
          confidence: { type: "number", description: "Confidence score between 0 and 1" }
        },
        required: ["content"]
      },
      execute: async (params: MemoryPayload) => {
        onActivityLog?.(`[WebMCP] Invoking store_context: "${params.content}"`);
        const res = await fetch("/api/memory", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            ...params,
            source: sourceName,
            confidence: params.confidence ?? 0.95
          })
        });
        const data = await res.json();
        onActivityLog?.(`[Tether] Stored memory ID: ${data.id} (Status: 201)`);
        return data;
      }
    });

    // Register retrieve_context tool
    modelContext.registerTool({
      name: "retrieve_context",
      description: "Retrieve relevant user context from Tether for the current task.",
      inputSchema: {
        type: "object",
        properties: {
          query: { type: "string", description: "Context search query" },
          source_filter: { type: "string", description: "Optional source filter" }
        },
        required: ["query"]
      },
      execute: async ({ query, source_filter }: { query: string; source_filter?: string }) => {
        onActivityLog?.(`[WebMCP] Invoking retrieve_context("${query}")`);
        const url = `/api/memory/search?q=${encodeURIComponent(query)}${source_filter ? `&source=${source_filter}` : ""}`;
        const res = await fetch(url);
        const results = await res.json();
        onActivityLog?.(`[Tether] Retrieved ${results.length} memories for "${query}"`);
        return { memories: results };
      }
    });
  }, [sourceName, onActivityLog]);

  return { isSupported };
}
```
